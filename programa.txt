// Declaracoes: lista de variaveis e inicializacao
int a = 10, b, c;
float x = 2.5E+1, y = 0.5;

read(b);

y = x / 2.0 - -1.5e-2;

// Precedencia: || < && < (== !=) < (< > <= >=) < (+ -) < (* / %) < (! -unario)
if (a == b < c + 1 && !(a > 0) || c != 0)
    print(a);
else
    print(b);

// Dangling else: o else pertence ao if mais interno
if (a > 0)
    if (b > 0)
        print(a + b);
    else
        print(a - b);

// while com bloco, declaracao interna e if sem else

    int i = 0;
    a = a - 1;
    if (a % 2 == 0) {
        print(a);
    }
}

{ float z = x * y; print(z); }     // bloco isolado
